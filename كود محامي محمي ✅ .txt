def clean_formatting(text):
    """ تنظيف النص من النجوم والتنسيقات النقطية """
    if not text:
        return ""
    text = text.replace('**', '').replace('*', '')
    text = text.replace('###', '').replace('##', '').replace('#', '')
    return text

def call_ai_api(prompt_text):
    clean_key = MY_SAVED_API_KEY.strip()
    url = f"https://generativelanguage.googleapis.com/v1beta/models/gemini-3.6-flash:generateContent?key={clean_key}"
    
    headers = {'Content-Type': 'application/json'}
    payload = {
        "contents": [{
            "parts": [{"text": prompt_text}]
        }]
    }
    
    for attempt in range(3):
        try:
            res = requests.post(url, headers=headers, json=payload, timeout=90, verify=False)
            if res.status_code == 200:
                data = res.json()
                raw_text = data['candidates'][0]['content']['parts'][0]['text']
                return clean_formatting(raw_text)
            elif res.status_code == 404:
                return f"خطأ: النموذج غير متاح، يرجى مراجعة مفتاح الـ API."
        except requests.exceptions.Timeout:
            if attempt < 2:
                time.sleep(2)
                continue
            return "فشل الاتصال: بطء شديد في تغطية الإنترنت بالهاتف."
        except Exception as e:
            return f"فشل الاتصال بالشبكة: {str(e)}"
            
    return "تعذر استلام الرد، يرجى التأكد من استقرار شبكة الإنترنت."

@app.route('/api/generate_draft', methods=['POST'])
def generate_draft():
    data = request.json
    doc_type = data.get('doc_type', 'دعوى قضائية')
    subject = data.get('subject', '')
    details = data.get('details', '')

    prompt = f"""
أنت مستشار وصائغ قانوني خبير في القضاء والتشريعات اليمنية.
قم بصياغة {doc_type} بناءً على المعطيات التالية:
- الموضوع/العنوان: {subject}
- التفاصيل والوقائع المقدمة: {details}

شروط مهمة جداً لضمان الدقة:
1. توخي الدقة التامة واليقين عند استشهادك بأرقام المواد من التشريعات اليمنية النافذة (مثل: قانون الإجراءات المدنية والتنفيذ، القانون المدني، قانون الإثبات).
2. إذا لم تكن متأكداً 100% من رقم المادة بالضبط، اذكر نص المادة القانونية واسم القانون فقط دون التخمين برقم خاطئ.
3. اكتب النص بلغة عربية رسمية ونظيفة تماماً دون استخدام أي رموز مثل * أو **.
"""
    result = call_ai_api(prompt)
    return jsonify({"status": "success", "result": result})

@app.route('/api/analyze', methods=['POST'])
def analyze():
    data = request.json
    content = data.get('content', '')

    prompt = f"قم بإجراء تحليل ودراسة واستشارة قانونية دقيقة وفق القانون اليمني للنص التالي بدون استخدام أي نجوم أو رموز تنسيق:\n\n{content}"
    result = call_ai_api(prompt)
    return jsonify({"status": "success", "analysis": result})

@app.route('/api/export_word', methods=['POST'])
def export_word():
    data = request.json
    text = data.get('text', '')
    
    doc = Document()
    doc.add_heading('عريضة قانونية', 0)
    for paragraph in text.split('\n'):
        doc.add_paragraph(paragraph)
    
    file_stream = io.BytesIO()
    doc.save(file_stream)
    file_stream.seek(0)
    
    return send_file(
        file_stream, 
        as_attachment=True, 
        download_name="عريضة_قانونية.docx", 
        mimetype="application/vnd.openxmlformats-officedocument.wordprocessingml.document"
    )
